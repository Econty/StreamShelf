cmake_minimum_required(VERSION 3.22)
project(StreamShelfNative VERSION 2.3.0 LANGUAGES CXX)

include(CTest)
option(STREAMSHELF_STATIC_RUNTIME "Link the C++ runtime statically" ON)
option(STREAMSHELF_NATIVE_GUIDE_TESTS "Build interactive desktop guide tests" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  if(STREAMSHELF_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
  endif()
  add_compile_options(/utf-8 /EHsc)
endif()

add_library(streamshelf_core STATIC src/core.cpp)
target_include_directories(streamshelf_core PUBLIC src vendor)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE streamshelf_core)
  add_test(NAME native_core COMMAND core_tests)
  add_executable(guide_flow_tests tests/guide_flow_resume_tests.cpp)
  target_include_directories(guide_flow_tests PRIVATE src)
  add_test(NAME guide_flow_resume COMMAND guide_flow_tests)
  foreach(name probe_result assembly_batch download_validation)
    add_executable(${name}_tests tests/${name}_tests.cpp)
    target_link_libraries(${name}_tests PRIVATE streamshelf_core)
    add_test(NAME ${name} COMMAND ${name}_tests)
  endforeach()
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    foreach(name connector-page extension media-discovery)
      add_test(NAME js_${name} COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}-tests.mjs)
      set_tests_properties(js_${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    endforeach()
  else()
    message(WARNING "Node.js not found; extension tests are unavailable")
  endif()
endif()

if(WIN32)
  enable_language(RC)
  add_library(streamshelf_windows INTERFACE)
  target_compile_definitions(streamshelf_windows INTERFACE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(streamshelf_windows INTERFACE streamshelf_core opengl32 gdi32 user32 shell32 ole32 oleaut32 uuid uiautomationcore version winhttp bcrypt crypt32 advapi32 comdlg32 shlwapi windowscodecs)
  if(MINGW AND STREAMSHELF_STATIC_RUNTIME)
    target_link_options(streamshelf_windows INTERFACE -static -static-libgcc -static-libstdc++)
  endif()
  # Shared by the application and tests; each translation unit is compiled once.
  add_library(streamshelf_platform STATIC src/platform.cpp src/component_manager.cpp)
  target_link_libraries(streamshelf_platform PUBLIC streamshelf_windows)
  add_library(streamshelf_connector STATIC src/connector.cpp)
  target_link_libraries(streamshelf_connector PUBLIC streamshelf_platform)
  add_library(streamshelf_render STATIC src/render.cpp)
  target_link_libraries(streamshelf_render PUBLIC streamshelf_platform)
  add_library(streamshelf_installer STATIC src/extension_installer.cpp)
  target_link_libraries(streamshelf_installer PUBLIC streamshelf_render)
  add_library(streamshelf_preview STATIC src/preview.cpp)
  target_link_libraries(streamshelf_preview PUBLIC streamshelf_platform)
  add_executable(StreamShelf WIN32 src/main.cpp src/app.cpp src/assembly.cpp src/assembly_download.cpp src/local_media.cpp src/constructor_ui.cpp src/ui.cpp src/app.rc)
  target_include_directories(StreamShelf PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(StreamShelf PRIVATE streamshelf_connector streamshelf_installer streamshelf_preview)
  if(MINGW)
    target_link_options(StreamShelf PRIVATE -municode)
  endif()
  add_custom_command(TARGET StreamShelf POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:StreamShelf>/Assets"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/extension" "$<TARGET_FILE_DIR:StreamShelf>/extensions")

  if(BUILD_TESTING)
    add_executable(component_manager_tests tests/component_manager_tests.cpp)
    target_link_libraries(component_manager_tests PRIVATE streamshelf_platform)
    add_test(NAME component_manager COMMAND component_manager_tests)
    add_executable(terminal_tests tests/terminal_windows_tests.cpp)
    target_link_libraries(terminal_tests PRIVATE streamshelf_platform)
    add_test(NAME terminal_path COMMAND terminal_tests)
    add_executable(connector_tests tests/connector_windows_tests.cpp)
    target_link_libraries(connector_tests PRIVATE streamshelf_connector)
    add_test(NAME connector_duplex_and_timeout COMMAND connector_tests)
    set_tests_properties(connector_duplex_and_timeout PROPERTIES TIMEOUT 30)
    add_executable(container_plan_tests tests/container_plan_tests.cpp)
    target_link_libraries(container_plan_tests PRIVATE streamshelf_platform)
    add_test(NAME container_plan COMMAND container_plan_tests)
    add_executable(preview_tests tests/preview_tests.cpp)
    target_link_libraries(preview_tests PRIVATE streamshelf_preview)
    add_test(NAME preview_decode COMMAND preview_tests)
    set_tests_properties(preview_decode PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(extension_installer_tests tests/extension_installer_tests.cpp)
    target_link_libraries(extension_installer_tests PRIVATE streamshelf_installer)
    add_test(NAME extension_installer COMMAND extension_installer_tests)
    if(STREAMSHELF_NATIVE_GUIDE_TESTS)
      add_executable(extension_installer_native_tests tests/extension_installer_native_tests.cpp src/extension_installer.cpp)
      target_compile_definitions(extension_installer_native_tests PRIVATE STREAMSHELF_INSTALLER_TESTING)
      target_link_libraries(extension_installer_native_tests PRIVATE streamshelf_render)
      add_test(NAME extension_installer_desktop COMMAND extension_installer_native_tests)
      add_custom_command(TARGET extension_installer_native_tests POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:extension_installer_native_tests>/Assets")
      set_tests_properties(extension_installer_desktop PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 100 LABELS desktop)
    endif()
  endif()
  install(TARGETS StreamShelf RUNTIME DESTINATION .)
  install(DIRECTORY Assets DESTINATION .)
  install(DIRECTORY extension/ DESTINATION extensions PATTERN "firefox-development" EXCLUDE PATTERN "StreamShelf-Connector-1.0.2.xpi" EXCLUDE)
  install(FILES README.md CHANGES-2.3.md BUILDING.md VERIFICATION.md THIRD_PARTY.md BROWSER-INSTALLER.md BROWSER-INSTALLER-VERIFICATION.md DESTINATION .)
  install(DIRECTORY licenses DESTINATION .)
endif()
