  # This is the common release packaging boundary, including direct
  # `cmake --install` calls and CI. Development installs remain available while
  # the current Firefox build is awaiting Mozilla's signature.
  install(CODE "
    # A single-config release tree still contains release binaries if callers
    # pass --config Debug to cmake --install. Check both the requested install
    # configuration and the configuration that produced the binaries.
    string(TOUPPER \"\${CMAKE_INSTALL_CONFIG_NAME}\" _streamshelf_install_config)
    string(TOUPPER \"${CMAKE_BUILD_TYPE}\" _streamshelf_build_config)
    if(${STREAMSHELF_BETA_WITHOUT_FIREFOX})
      message(STATUS \"Building an explicit beta package without the signed Firefox connector\")
    elseif(_streamshelf_install_config MATCHES \"^(RELEASE|RELWITHDEBINFO|MINSIZEREL)$\" OR
           _streamshelf_build_config MATCHES \"^(RELEASE|RELWITHDEBINFO|MINSIZEREL)$\")
      find_program(_streamshelf_python NAMES python3 python)
      if(NOT _streamshelf_python)
        message(FATAL_ERROR \"Python 3 is required to verify the signed Firefox package before a release install\")
      endif()
      execute_process(COMMAND \"\${_streamshelf_python}\" \"${CMAKE_CURRENT_SOURCE_DIR}/scripts/Verify-FirefoxPackage.py\"
        --repo \"${CMAKE_CURRENT_SOURCE_DIR}\"
        --package \"${CMAKE_CURRENT_SOURCE_DIR}/extension/firefox-signed/StreamShelf-Connector-1.0.5.xpi\"
        RESULT_VARIABLE _streamshelf_firefox_result)
      if(NOT \"\${_streamshelf_firefox_result}\" STREQUAL \"0\")
        message(FATAL_ERROR \"Release install requires the current Mozilla-signed Firefox connector; see FIREFOX-SIGNING.md\")
      endif()
    endif()
  ")

  endif()
