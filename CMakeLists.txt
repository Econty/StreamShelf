cmake_minimum_required(VERSION 3.22)
project(StreamShelfNative VERSION 2.2.1 LANGUAGES CXX)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(streamshelf_core src/core.cpp)
target_include_directories(streamshelf_core PUBLIC src vendor)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE streamshelf_core)
enable_testing()
add_test(NAME native_core COMMAND core_tests)
add_executable(guide_flow_tests tests/guide_flow_resume_tests.cpp)
target_include_directories(guide_flow_tests PRIVATE src)
add_test(NAME guide_flow_resume COMMAND guide_flow_tests)
foreach(name probe_result assembly_batch download_validation)
 add_executable(${name}_tests tests/${name}_tests.cpp)
 target_link_libraries(${name}_tests PRIVATE streamshelf_core)
 add_test(NAME ${name} COMMAND ${name}_tests)
endforeach()
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
 foreach(name connector-page extension media-discovery)
  add_test(NAME js_${name} COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}-tests.mjs)
 endforeach()
endif()
if(WIN32)
 enable_language(RC)
 add_executable(StreamShelf WIN32 src/main.cpp src/platform.cpp src/preview.cpp src/connector.cpp src/extension_installer.cpp src/render.cpp src/app.cpp src/assembly.cpp src/assembly_download.cpp src/local_media.cpp src/constructor_ui.cpp src/ui.cpp src/app.rc)
 target_link_libraries(StreamShelf PRIVATE streamshelf_core opengl32 gdi32 user32 shell32 ole32 oleaut32 uuid uiautomationcore version winhttp bcrypt crypt32 advapi32 comdlg32 shlwapi windowscodecs)
 target_compile_definitions(StreamShelf PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_include_directories(StreamShelf PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 add_executable(terminal_tests tests/terminal_windows_tests.cpp src/platform.cpp)
 target_link_libraries(terminal_tests PRIVATE streamshelf_core shell32 ole32 uuid winhttp bcrypt crypt32 advapi32 comdlg32 shlwapi user32)
 target_compile_definitions(terminal_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 add_test(NAME terminal_path COMMAND terminal_tests)
 add_executable(connector_tests tests/connector_windows_tests.cpp src/platform.cpp src/connector.cpp)
 target_link_libraries(connector_tests PRIVATE streamshelf_core shell32 ole32 uuid winhttp bcrypt crypt32 advapi32 comdlg32 shlwapi user32)
 target_compile_definitions(connector_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 add_test(NAME connector_duplex_and_timeout COMMAND connector_tests)
 add_executable(extension_installer_tests tests/extension_installer_tests.cpp src/extension_installer.cpp src/platform.cpp src/render.cpp)
 target_link_libraries(extension_installer_tests PRIVATE streamshelf_core opengl32 gdi32 user32 shell32 ole32 oleaut32 uuid uiautomationcore version winhttp bcrypt crypt32 advapi32 comdlg32 shlwapi)
 target_compile_definitions(extension_installer_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 add_test(NAME extension_installer COMMAND extension_installer_tests)
 option(STREAMSHELF_NATIVE_GUIDE_TESTS "Run interactive desktop guide regression fixture" OFF)
 if(STREAMSHELF_NATIVE_GUIDE_TESTS)
  add_executable(extension_installer_native_tests tests/extension_installer_native_tests.cpp src/extension_installer.cpp src/platform.cpp src/render.cpp)
  target_link_libraries(extension_installer_native_tests PRIVATE streamshelf_core opengl32 gdi32 user32 shell32 ole32 oleaut32 uuid uiautomationcore version winhttp bcrypt crypt32 advapi32 comdlg32 shlwapi)
  target_compile_definitions(extension_installer_native_tests PRIVATE STREAMSHELF_INSTALLER_TESTING UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_test(NAME extension_installer_desktop COMMAND extension_installer_native_tests)
  add_custom_command(TARGET extension_installer_native_tests POST_BUILD
   COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:extension_installer_native_tests>/Assets")
  set_tests_properties(extension_installer_desktop PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 100 LABELS desktop)
  if(NOT MSVC)
   target_link_options(extension_installer_native_tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
 endif()
 set_tests_properties(connector_duplex_and_timeout PROPERTIES TIMEOUT 20)
 if(MSVC)
  target_compile_options(StreamShelf PRIVATE /utf-8 /W4 /EHsc)
  target_compile_options(streamshelf_core PRIVATE /utf-8 /W4)
  set_property(TARGET StreamShelf streamshelf_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 else()
  target_link_options(StreamShelf PRIVATE -municode -static -static-libgcc -static-libstdc++)
  target_link_options(connector_tests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(terminal_tests PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_options(extension_installer_tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 install(TARGETS StreamShelf RUNTIME DESTINATION .)
 install(DIRECTORY Assets DESTINATION .)
 install(DIRECTORY extension/ DESTINATION extensions)
 install(FILES README.md VERIFICATION.md THIRD_PARTY.md BROWSER-INSTALLER.md BROWSER-INSTALLER-VERIFICATION.md DESTINATION .)
 install(DIRECTORY licenses DESTINATION .)
endif()
